Exercice 1 : Vérification de nombre (positif/négatif/nul)

Code JavaScript avec commentaires :

function verifierNombre() {
    // 1. Récupération de la valeur saisie
    // getElementById() récupère l'élément HTML
    // .value obtient la valeur du champ
    // parseFloat() convertit le texte en nombre décimal
    const nombre = parseFloat(document.getElementById('nombre').value);
    
    // 2. Validation : vérifier si l'utilisateur a bien saisi un nombre
    // isNaN() retourne true si ce n'est PAS un nombre
    if (isNaN(nombre)) {
        document.getElementById('result').innerHTML = ' Veuillez entrer un nombre valide';
        return; // On arrête la fonction ici
    }
    
    // 3. Structure conditionnelle avec if / else if / else
    // L'ordre est IMPORTANT : on teste d'abord > 0, puis < 0, sinon c'est 0
    let message;
    
    if (nombre > 0) {
        // Si le nombre est strictement supérieur à 0
        message = ` ${nombre} est un nombre POSITIF`;
    } else if (nombre < 0) {
        // Si le nombre est strictement inférieur à 0
        message = ` ${nombre} est un nombre NÉGATIF`;
    } else {
        // Dans tous les autres cas (donc si nombre === 0)
        message = ` ${nombre} est NUL (ni positif, ni négatif)`;
    }
    
    // 4. Affichage du résultat
    document.getElementById('result').innerHTML = message;
}
