Exercice 5 : Système d'accès sécurisé (opérateurs && et ||)










Code JavaScript avec commentaires :

function verifierAcces() {
    // 1. Récupération de toutes les données
    const username = document.getElementById('username').value.trim();
    const password = document.getElementById('password').value;
    const age = parseInt(document.getElementById('age').value);
    
    // Pour les checkbox, on utilise .checked qui retourne true ou false
    const estVIP = document.getElementById('vip').checked;
    const estAdmin = document.getElementById('admin').checked;
    
    // 2. Validation de base (champs requis)
    if (username === '' || password === '' || isNaN(age)) {
        document.getElementById('result').innerHTML = '❌ Veuillez remplir tous les champs';
        return;
    }
    
    // 3. Définition des critères d'accès
    // === vérifie l'égalité stricte (valeur ET type)
    // !== vérifie la différence stricte
    const usernameValide = username === 'admin' || username === 'user';
    const passwordValide = password.length >= 6;
    const ageValide = age >= 18;
    
    // 4. OPÉRATEURS LOGIQUES : && (ET) et || (OU)
    
    // OPÉRATEUR && (ET LOGIQUE)
    // Retourne true SEULEMENT SI TOUTES les conditions sont vraies
    // Si une seule est false, le résultat est false
    // Exemples :
    // true && true && true = true
    // true && false && true = false
    
    // OPÉRATEUR || (OU LOGIQUE)
    // Retourne true SI AU MOINS UNE condition est vraie
    // Retourne false seulement si TOUTES sont fausses
    // Exemples :
    // true || false = true
    // false || false = false
    // true || true = true
    
    // 5. Règles d'accès complexes avec combinaison de && et ||
    
    // Règle 1 : Accès ADMIN (priorité absolue)
    // Un administrateur a TOUJOURS accès, peu importe les autres critères
    if (estAdmin) {
        document.getElementById('result').innerHTML = 
            '✅ ACCÈS ACCORDÉ
Statut : Administrateur (accès total)';
        return;
    }
    
    // Règle 2 : Accès VIP
    // Un membre VIP doit avoir :
    // - username valide ET password valide ET âge >= 18
    // Les opérateurs && s'évaluent de gauche à droite
    if (estVIP && usernameValide && passwordValide && ageValide) {
        document.getElementById('result').innerHTML = 
            '✅ ACCÈS ACCORDÉ
Statut : Membre VIP';
        return;
    }
    
    // Règle 3 : Accès Standard
    // Un utilisateur standard doit avoir TOUS les critères de base
    // ET ne pas être mineur
    if (usernameValide && passwordValide && ageValide) {
        document.getElementById('result').innerHTML = 
            '✅ ACCÈS ACCORDÉ
Statut : Utilisateur standard';
        return;
    }
    
    // 6. Si aucune règle n'est satisfaite, on refuse l'accès
    // et on détaille les problèmes
    let problemes = [];
    
    if (!usernameValide) {
        problemes.push('❌ Nom d\'utilisateur incorrect (doit être "admin" ou "user")');
    }
    if (!passwordValide) {
        problemes.push('❌ Mot de passe trop court (minimum 6 caractères)');
    }
    if (!ageValide) {
        problemes.push('❌ Âge insuffisant (minimum 18 ans)');
    }
    
    const message = `
        ❌ ACCÈS REFUSÉ


        Problèmes détectés :

        ${problemes.join('
')}
        


        Astuce : Les membres VIP ou Administrateurs ont des privilèges spéciaux
    `;
    
    document.getElementById('result').innerHTML = message;
}
