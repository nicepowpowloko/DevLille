01

C'est quoi Express ?

Fondamentaux

Express.js est un framework minimaliste pour Node.js. Il ajoute une couche d'abstraction sur le module HTTP natif pour gérer facilement le routing, les middlewares et les réponses HTTP.

🔍
Rappel : sans Express, créer une route GET /users demandait de parser l'URL manuellement dans des if/else. Express réduit ça à une ligne : app.get('/users', handler).
Routing
Associe une URL + méthode HTTP à une fonction handler. Supporte les paramètres dynamiques, les query strings et les regex.
Middleware
Fonctions qui s'exécutent dans la chaîne de traitement d'une requête. Parsing JSON, authentification, logs, CORS…
Écosystème
Des centaines de middlewares NPM prêts à l'emploi : cors, helmet, express-validator, multer, passport…
Non-opinioné
Express ne t'impose pas de structure de projet. Tu choisis ton ORM, ta structure de dossiers, ton système d'authentification…

import express from 'express';
import 'dotenv/config';

const app  = express();
const PORT = process.env.PORT || 3000;

// Middleware pour parser le JSON des requêtes
app.use(express.json());

// Première route
app.get('/', (req, res) => {
    res.json({ message: 'API opérationnelle' });
});

// Démarrer le serveur
app.listen(PORT, () => {
    console.log(`Serveur Express sur http://localhost:${PORT}`);
});

Routes de base — GET, POST, PUT, DELETE

CRUD

Chaque route associe une méthode HTTP + un chemin URL à une fonction handler qui reçoit req (requête) et res (réponse).

MéthodeRouteActionCode HTTP
GET/usersLire tous les utilisateurs200
GET/users/:idLire un utilisateur par id200 / 404
POST/usersCréer un utilisateur201
PUT/users/:idRemplacer un utilisateur200
PATCH/users/:idModifier partiellement200
DELETE/users/:idSupprimer un utilisateur204
routes/users.js — CRUD complet (données en mémoire)
import express from 'express';
const router = express.Router();

// Données simulées (en prod → base de données)
let users = [
    { id: 1, nom: 'Alice', email: 'alice@mail.fr' },
    { id: 2, nom: 'Bob',   email: 'bob@mail.fr'   },
];

// GET /users — tous les utilisateurs
router.get('/', (req, res) => {
    res.json(users);
});

// GET /users/:id — un utilisateur
router.get('/:id', (req, res) => {
    const user = users.find(u => u.id === Number(req.params.id));
    if (!user) return res.status(404).json({ erreur: 'Utilisateur introuvable' });
    res.json(user);
});

// POST /users — créer
router.post('/', (req, res) => {
    const { nom, email } = req.body;
    if (!nom || !email) {
        return res.status(400).json({ erreur: 'nom et email requis' });
    }
    const newUser = { id: Date.now(), nom, email };
    users.push(newUser);
    res.status(201).json(newUser);
});

// PUT /users/:id — remplacer
router.put('/:id', (req, res) => {
    const idx = users.findIndex(u => u.id === Number(req.params.id));
    if (idx === -1) return res.status(404).json({ erreur: 'Introuvable' });
    users[idx] = { id: Number(req.params.id), ...req.body };
    res.json(users[idx]);
});

// DELETE /users/:id — supprimer
router.delete('/:id', (req, res) => {
    const before = users.length;
    users = users.filter(u => u.id !== Number(req.params.id));
    if (users.length === before) return res.status(404).json({ erreur: 'Introuvable' });
    res.status(204).send();  // 204 = No Content
});

export default router;

Paramètres de route & Query strings

Express — req.params, req.query, req.body
//  Paramètres de route (:id) ─
// URL : /users/42
app.get('/users/:id', (req, res) => {
    console.log(req.params.id);     // '42' (string)
    console.log(Number(req.params.id)); // 42  (number)
});

// Plusieurs paramètres
// URL : /users/5/articles/12
app.get('/users/:userId/articles/:articleId', (req, res) => {
    const { userId, articleId } = req.params;
    res.json({ userId, articleId });
});

// ── Query strings (après ?) ──
// URL : /articles?page=2&limit=10&search=node
app.get('/articles', (req, res) => {
    const page   = Number(req.query.page)   || 1;
    const limit  = Number(req.query.limit)  || 10;
    const search = req.query.search || '';
    res.json({ page, limit, search });
});

// ── Body (POST/PUT/PATCH) ──
// Requête : POST /users avec body JSON
app.post('/users', (req, res) => {
    const { nom, email } = req.body;  // nécessite express.json()
    res.status(201).json({ nom, email });
});

// ── Headers ──
const token = req.headers['authorization'];
const lang  = req.get('Accept-Language');

Router — organiser les routes

Quand une API grandit, on sépare les routes dans des fichiers dédiés avec express.Router(). Chaque ressource a son propre fichier de routes.

src/app.js — monter les routers
import express      from 'express';
import usersRouter  from './routes/users.js';
import articlesRouter from './routes/articles.js';

const app = express();
app.use(express.json());

// Monter les routers avec leur préfixe
app.use('/api/users',    usersRouter);
app.use('/api/articles', articlesRouter);

// GET /api/users    → usersRouter.get('/')
// GET /api/users/5  → usersRouter.get('/:id')
// POST /api/articles → articlesRouter.post('/')

app.listen(3000);

Middleware — le concept clé d'Express

Architecture

Un middleware est une fonction avec la signature (req, res, next). Il peut modifier req et res, terminer la requête, ou appeler next() pour passer au suivant.

// Chaîne de traitement d'une requête
Requête
→
logger()
→
auth()
→
validate()
→
handler
→
Réponse
middleware — exemples pratiques
// ── Middleware global (toutes les routes) ──
const logger = (req, res, next) => {
    console.log(`${new Date().toISOString()} ${req.method} ${req.url}`);
    next(); // ← OBLIGATOIRE pour passer au suivant
};
app.use(logger);

// ── Middleware d'authentification JWT ──
const authMiddleware = (req, res, next) => {
    const token = req.headers.authorization?.replace('Bearer ', '');
    if (!token) {
        return res.status(401).json({ erreur: 'Token manquant' });
        // ← on NE call pas next(), la chaîne s'arrête ici
    }
    // Vérifier le token (ex: jwt.verify)
    req.user = { id: 1, role: 'admin' }; // ajouter l'user à req
    next();
};

// Appliquer sur une route spécifique
app.get('/profile', authMiddleware, (req, res) => {
    res.json(req.user);
});

// Appliquer sur un groupe de routes
router.use(authMiddleware);  // protège toutes les routes du router

// ── Middleware de validation ──
const validateUser = (req, res, next) => {
    const { nom, email } = req.body;
    if (!nom?.trim())   return res.status(400).json({ erreur: 'nom requis' });
    if (!email?.includes('@')) return res.status(400).json({ erreur: 'email invalide' });
    next();
};
router.post('/', validateUser, usersController.create);

Middlewares intégrés & populaires

app.js — middlewares essentiels
import express from 'express';
import cors    from 'cors';      // npm i cors
import helmet  from 'helmet';    // npm i helmet
import morgan  from 'morgan';    // npm i morgan

const app = express();

// ── Sécurité — headers HTTP (indispensable en prod)
app.use(helmet());

// ── CORS — autoriser le front à appeler l'API
app.use(cors({
    origin: 'http://localhost:5173',  // URL du front React/Vite
    methods: ['GET','POST','PUT','DELETE'],
    credentials: true,
}));

// ── Logs des requêtes (dev = format coloré)
app.use(morgan('dev'));

// ── Parser le JSON des requêtes
app.use(express.json());

// ── Parser les formulaires HTML (application/x-www-form-urlencoded)
app.use(express.urlencoded({ extended: true }));

// ── Servir des fichiers statiques
app.use(express.static('public')); // dossier public/

Gestion centralisée des erreurs

Express a un mécanisme de gestion d'erreurs via un middleware à 4 paramètres (err, req, res, next) placé en dernier.

app.js — gestion d'erreurs et route 404
// ── Propager une erreur depuis une route ──
app.get('/users/:id', async (req, res, next) => {
    try {
        const user = await UserService.findById(req.params.id);
        if (!user) {
            const err = new Error('Utilisateur introuvable');
            err.status = 404;
            return next(err);  // passe l'erreur au gestionnaire
        }
        res.json(user);
    } catch (err) {
        next(err);  // toutes les erreurs non gérées
    }
});

// ── 404 — route non trouvée (avant le gestionnaire d'erreurs)
app.use((req, res) => {
    res.status(404).json({ erreur: `Route ${req.method} ${req.url} introuvable` });
});

// ── Gestionnaire d'erreurs global (4 params — obligatoire) ──
app.use((err, req, res, next) => {
    console.error(err.stack);

    const status  = err.status  || 500;
    const message = err.message || 'Erreur interne du serveur';

    res.status(status).json({
        erreur: message,
        // stack uniquement en développement
        ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
    });
});
Dans les routes async, toujours entourer d'un try/catch et appeler next(err). Sinon les erreurs asynchrones ne sont pas capturées par le gestionnaire global.

API REST complète — Architecture MVC

Best practices

En production, on sépare les responsabilités : Routes (définir les endpoints), Controllers (logique métier), Services (accès aux données).

controllers/users.controller.js
import * as UserService from '../services/users.service.js';

export const getAll = async (req, res, next) => {
    try {
        const users = await UserService.findAll();
        res.json(users);
    } catch (err) { next(err); }
};

export const getById = async (req, res, next) => {
    try {
        const user = await UserService.findById(req.params.id);
        if (!user) return res.status(404).json({ erreur: 'Non trouvé' });
        res.json(user);
    } catch (err) { next(err); }
};

export const create = async (req, res, next) => {
    try {
        const newUser = await UserService.create(req.body);
        res.status(201).json(newUser);
    } catch (err) { next(err); }
};

export const pool = mysql.createPool({
    host:     process.env.DB_HOST,
    port:     process.env.DB_PORT,
    database: process.env.DB_NAME,
    user:     process.env.DB_USER,
    password: process.env.DB_PASS,
    waitForConnections: true,
    connectionLimit: 10,
});
services/users.service.js — requêtes SQL
import { pool } from '../db.js';

export const findAll = async () => {
    const [rows] = await pool.query('SELECT * FROM utilisateurs');
    return rows;
};

export const findById = async (id) => {
    // Requête préparée — protection injection SQL
    const [rows] = await pool.query(
        'SELECT * FROM utilisateurs WHERE id = ?', [id]
    );
    return rows[0] || null;
};

export const create = async ({ nom, email, motDePasse }) => {
    const [result] = await pool.query(
        'INSERT INTO utilisateurs (nom, email, mot_de_passe) VALUES (?, ?, ?)',
        [nom, email, motDePasse]
    );
    return { id: result.insertId, nom, email };
};

export const update = async (id, data) => {
    await pool.query(
        'UPDATE utilisateurs SET nom = ?, email = ? WHERE id = ?',
        [data.nom, data.email, id]
    );
    return findById(id);
};

export const remove = async (id) => {
    const [result] = await pool.query('DELETE FROM utilisateurs WHERE id = ?', [id]);
    return result.affectedRows > 0;
};
