Les TABLEAUX en JavaScript

Qu'est-ce qu'un tableau (Array) ?

Un tableau est une liste ordonnée qui peut contenir plusieurs valeurs dans une seule variable.

C'est comme un casier de rangement avec plusieurs compartiments numérotés.

Représentation visuelle d'un tableau
Index 0
"Alice"
Index 1
"Bob"
Index 2
"Charlie"
Index 3
"David"
let prenoms = ["Alice", "Bob", "Charlie", "David"];
RÈGLE IMPORTANTE :
Les tableaux commencent à l'index 0 (pas 1) !
• Premier élément = index 0
• Deuxième élément = index 1
• Troisième élément = index 2

Créer un tableau

Tableau vide

// Méthode 1 : Crochets [] let monTableau = []; // Méthode 2 : new Array() let monTableau = new Array();
Astuce : Préférez la méthode avec [], c'est plus simple et plus rapide !

Tableau avec valeurs

// Nombres let nombres = [10, 20, 30, 40]; // Textes let fruits = ["Pomme", "Banane", "Orange"]; // Mixte let mixte = ["Alice", 25, true, "Paris"];

Accéder aux éléments

Lire un élément avec son index
Index 0
"Pomme"
Index 1
"Banane"
Index 2
"Orange"
let fruits = ["Pomme", "Banane", "Orange"]; // Accéder à un élément : nomTableau[index] console.log(fruits[0]); // Affiche : Pomme console.log(fruits[1]); // Affiche : Banane console.log(fruits[2]); // Affiche : Orange // Premier élément let premier = fruits[0]; console.log(premier); // Affiche : Pomme // Dernier élément let dernier = fruits[fruits.length - 1]; console.log(dernier); // Affiche : Orange
FORMULE MAGIQUE :
Pour accéder au dernier élément d'un tableau : tableau[tableau.length - 1]
Pourquoi -1 ? Parce que les index commencent à 0 !

Propriétés importantes

length - Connaître la taille du tableau
let fruits = ["Pomme", "Banane", "Orange"]; // Obtenir le nombre d'éléments console.log(fruits.length); // Affiche : 3 // Vérifier si le tableau est vide if (fruits.length === 0) { console.log("Le tableau est vide"); } else { console.log("Le tableau contient " + fruits.length + " éléments"); }
RAPPEL : length donne le NOMBRE d'éléments, PAS le dernier index !
Tableau de 3 éléments → length = 3, mais dernier index = 2

Méthodes pour AJOUTER des éléments

push() - Ajouter à la FIN
Alice
Bob
Charlie ← NOUVEAU
let prenoms = ["Alice", "Bob"]; // Ajouter "Charlie" à la fin prenoms.push("Charlie"); console.log(prenoms); // Affiche : ["Alice", "Bob", "Charlie"] // On peut ajouter plusieurs éléments prenoms.push("David", "Emma");
unshift() - Ajouter au DÉBUT
Zoé ← NOUVEAU
Alice
Bob
let prenoms = ["Alice", "Bob"]; // Ajouter "Zoé" au début prenoms.unshift("Zoé"); console.log(prenoms); // Affiche : ["Zoé", "Alice", "Bob"]

Méthodes pour SUPPRIMER des éléments

pop() - Supprimer le DERNIER
Alice
Bob
Charlie ✗
let prenoms = ["Alice", "Bob", "Charlie"]; // Supprimer le dernier élément let supprime = prenoms.pop(); console.log(supprime); // Affiche : Charlie console.log(prenoms); // Affiche : ["Alice", "Bob"]
shift() - Supprimer le PREMIER
Alice ✗
Bob
Charlie
let prenoms = ["Alice", "Bob", "Charlie"]; // Supprimer le premier élément let supprime = prenoms.shift(); console.log(supprime); // Affiche : Alice console.log(prenoms); // Affiche : ["Bob", "Charlie"]
Schéma récapitulatif AJOUT/SUPPRESSION
unshift() [ éléments ] push()
shift() [ éléments ] pop()

Parcourir un tableau (Boucles)

Boucle FOR classique
let fruits = ["Pomme", "Banane", "Orange"]; // Parcourir avec for for (let i = 0; i < fruits.length; i++) { console.log("Index " + i + " : " + fruits[i]); } // Résultat : // Index 0 : Pomme // Index 1 : Banane // Index 2 : Orange
Explication :
• i = 0 : On commence à l'index 0
• i < fruits.length : On continue tant qu'on n'a pas dépassé
• i++ : On passe à l'élément suivant
forEach() - Méthode moderne
let fruits = ["Pomme", "Banane", "Orange"]; // Parcourir avec forEach fruits.forEach(function(fruit, index) { console.log("Index " + index + " : " + fruit); }); // Version fléchée (moderne) fruits.forEach((fruit, index) => { console.log(`Index ${index} : ${fruit}`); });
Astuce :
forEach() est plus simple car on n'a pas besoin de gérer l'index manuellement !

Rechercher dans un tableau

indexOf() - Trouver la position
let fruits = ["Pomme", "Banane", "Orange"]; // Chercher "Banane" let position = fruits.indexOf("Banane"); console.log(position); // Affiche : 1 // Chercher quelque chose qui n'existe pas let pos = fruits.indexOf("Kiwi"); console.log(pos); // Affiche : -1 (non trouvé)
includes() - Vérifier la présence
let fruits = ["Pomme", "Banane", "Orange"]; // Vérifier si "Banane" existe if (fruits.includes("Banane")) { console.log("Banane trouvée !"); } // Vérifier si "Kiwi" existe console.log(fruits.includes("Kiwi")); // false

Transformer un tableau

map() - Transformer chaque élément

map() parcourt le tableau et crée un nouveau tableau avec les valeurs transformées.

Tableau initial : [1, 2, 3, 4]
↓ map(nombre × 2)
Nouveau tableau : [2, 4, 6, 8]
let nombres = [1, 2, 3, 4]; // Multiplier chaque nombre par 2 let doubles = nombres.map(function(nombre) { return nombre * 2; }); console.log(doubles); // Affiche : [2, 4, 6, 8] // Version fléchée (moderne) let doubles = nombres.map(nombre => nombre * 2);
Exemple concret : Ajouter "€" à chaque prix
let prix = [10, 20, 30]; let prixAvecEuro = prix.map(p => p + "€"); console.log(prixAvecEuro); // ["10€", "20€", "30€"]
filter() - Filtrer les éléments

filter() crée un nouveau tableau contenant uniquement les éléments qui respectent une condition.

Tableau initial : [5, 12, 8, 20, 3]
↓ filter(nombre > 10)
Nouveau tableau : [12, 20]
let nombres = [5, 12, 8, 20, 3]; // Garder seulement les nombres > 10 let grands = nombres.filter(function(nombre) { return nombre > 10; }); console.log(grands); // Affiche : [12, 20] // Version fléchée let grands = nombres.filter(nombre => nombre > 10);
Exemple concret : Filtrer les utilisateurs majeurs
let ages = [15, 22, 17, 30, 16]; let majeurs = ages.filter(age => age >= 18); console.log(majeurs); // [22, 30]

Autres méthodes utiles

join()

Convertir en texte

let fruits = ["Pomme", "Banane"]; let texte = fruits.join(", "); console.log(texte); // "Pomme, Banane"
slice()

Extraire une portion

let fruits = ["A", "B", "C", "D"]; let portion = fruits.slice(1, 3); console.log(portion); // ["B", "C"]
reverse()

Inverser l'ordre

let nombres = [1, 2, 3]; nombres.reverse(); console.log(nombres); // [3, 2, 1]
sort()

Trier le tableau

let fruits = ["C", "A", "B"]; fruits.sort(); console.log(fruits); // ["A", "B", "C"]
concat()

Fusionner tableaux

let t1 = [1, 2]; let t2 = [3, 4]; let fusion = t1.concat(t2); console.log(fusion); // [1, 2, 3, 4]
find()

Trouver un élément

let nombres = [5, 12, 8]; let trouve = nombres.find(n => n > 10); console.log(trouve); // 12

Tableaux d'objets (très important !)

Exemple : Liste d'utilisateurs
// Créer un tableau d'objets let utilisateurs = [ { nom: "Alice", age: 25, ville: "Paris" }, { nom: "Bob", age: 30, ville: "Lyon" }, { nom: "Charlie", age: 22, ville: "Paris" } ]; // Accéder à un utilisateur console.log(utilisateurs[0]); // { nom: "Alice", age: 25, ville: "Paris" } // Accéder à une propriété spécifique console.log(utilisateurs[0].nom); // "Alice" console.log(utilisateurs[1].age); // 30 // Parcourir et afficher utilisateurs.forEach(function(user) { console.log(user.nom + " a " + user.age + " ans"); });
Filtrer et transformer des objets
// Filtrer : Utilisateurs de Paris seulement let parisiens = utilisateurs.filter(function(user) { return user.ville === "Paris"; }); console.log(parisiens); // [{ nom: "Alice", ... }, { nom: "Charlie", ... }] // Map : Récupérer seulement les noms let noms = utilisateurs.map(function(user) { return user.nom; }); console.log(noms); // ["Alice", "Bob", "Charlie"] // Trouver : Utilisateur qui a 30 ans let user30 = utilisateurs.find(function(user) { return user.age === 30; }); console.log(user30); // { nom: "Bob", age: 30, ville: "Lyon" }

Pièges courants à éviter

PIÈGE 1 : Index qui n'existe pas

let fruits = ["Pomme", "Banane"]; console.log(fruits[5]); // undefined

Toujours vérifier que l'index est valide !

PIÈGE 2 : Oublier le return dans map()

// MAUVAIS let doubles = nombres.map(n => { n * 2; // Oubli du return ! }); // BON let doubles = nombres.map(n => { return n * 2; });

PIÈGE 3 : Confondre length et index

let tab = ["A", "B", "C"]; console.log(tab.length); // 3 console.log(tab[tab.length]); // undefined ! console.log(tab[tab.length - 1]); // "C"

PIÈGE 4 : Modifier pendant forEach

// Dangereux ! fruits.forEach(function() { fruits.push("Nouveau"); // Boucle infinie ! });
