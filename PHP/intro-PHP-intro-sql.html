Fondamentaux

SQL et les bases de donnees relationnelles

Une base de donnees relationnelle organise les donnees en tables (comme des feuilles de tableur). Chaque table a des colonnes (champs) et des lignes (enregistrements). SQL permet de lire, ajouter, modifier et supprimer ces donnees.

Commande SQLOperation CRUDRole
SELECTReadLire des donnees
INSERTCreateAjouter une ligne
UPDATEUpdateModifier une ou plusieurs lignes
DELETEDeleteSupprimer une ou plusieurs lignes
CREATE TABLE—Creer la structure d'une table
i
SQL n'est pas sensible a la casse pour les mots-cles. SELECT = select = Select. Convention : les mots-cles SQL en MAJUSCULES, les noms de tables et colonnes en minuscules. Cela ameliore la lisibilite.
01
Structure

CREATE TABLE — definir la structure

CREATE TABLE — exemple completSQL
-- Creer la table utilisateurs
CREATE TABLE utilisateurs (
    id         INT AUTO_INCREMENT PRIMARY KEY,
    nom        VARCHAR(100) NOT NULL,
    email      VARCHAR(255) NOT NULL UNIQUE,
    mot_de_passe VARCHAR(255) NOT NULL,
    role       ENUM('user', 'admin') DEFAULT 'user',
    cree_le    DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Creer la table produits
CREATE TABLE produits (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    nom         VARCHAR(200) NOT NULL,
    prix        DECIMAL(10, 2) NOT NULL,
    stock       INT DEFAULT 0,
    categorie_id INT,
    FOREIGN KEY (categorie_id) REFERENCES categories(id)
);

02
Read

SELECT — lire des donnees

SELECT — du simple au complexeSQL
-- Toutes les colonnes, toutes les lignes
SELECT * FROM utilisateurs;

-- Colonnes specifiques
SELECT id, nom, email FROM utilisateurs;

-- Filtrer avec WHERE
SELECT * FROM utilisateurs WHERE role = 'admin';
SELECT * FROM produits     WHERE prix < 50;
SELECT * FROM produits     WHERE stock > 0 AND prix < 100;

-- Trier les resultats
SELECT * FROM produits ORDER BY prix ASC;   -- croissant
SELECT * FROM produits ORDER BY prix DESC;  -- decroissant

-- Limiter le nombre de resultats (pagination)
SELECT * FROM produits LIMIT 10;
SELECT * FROM produits LIMIT 10 OFFSET 20;  -- page 3

-- Compter les lignes
SELECT COUNT(*) FROM utilisateurs;
SELECT COUNT(*) FROM produits WHERE stock = 0;

-- Fonctions d'agregation
SELECT AVG(prix), MIN(prix), MAX(prix), SUM(prix) FROM produits;

-- Recherche partielle avec LIKE
SELECT * FROM produits WHERE nom LIKE '%shirt%';
-- % = n'importe quelle sequence de caracteres

Create

INSERT — ajouter des donnees

INSERT INTO — inserer des lignesSQL
-- Inserer une ligne (toutes les colonnes sauf id)
INSERT INTO utilisateurs (nom, email, mot_de_passe)
VALUES ('Alice Dupont', 'alice@ex.com', 'hash_du_mdp');

-- Inserer avec valeurs par defaut
INSERT INTO produits (nom, prix)
VALUES ('T-shirt blanc', 19.99);
-- stock = 0 (valeur DEFAULT), categorie_id = NULL

-- Inserer plusieurs lignes en une requete
INSERT INTO produits (nom, prix, stock)
VALUES
    ('Jean slim',  49.99, 30),
    ('Veste cuir', 89.99, 15),
    ('Casquette',  14.99, 50);

-- Recuperer l'ID genere (en PHP avec PDO : lastInsertId())
SELECT LAST_INSERT_ID();
04
Update

UPDATE — modifier des donnees

UPDATE — modifier une ou plusieurs lignesSQL
-- Modifier un utilisateur specifique (toujours avec WHERE !)
UPDATE utilisateurs
SET nom = 'Alice Martin',
    email = 'alice.martin@ex.com'
WHERE id = 3;

-- Modifier un seul champ
UPDATE produits
SET prix = 14.99
WHERE id = 7;

-- Modifier plusieurs lignes a la fois
UPDATE produits
SET prix = prix * 0.90  -- -10% sur tous les prix
WHERE categorie_id = 2;

-- Incrementer une valeur
UPDATE produits
SET stock = stock - 1
WHERE id = 5;
!
UPDATE sans WHERE met a jour TOUTES les lignes. UPDATE produits SET prix = 0 met le prix a 0 pour tous les produits. Toujours verifier le WHERE avant d'executer un UPDATE. Bonne pratique : tester avec SELECT d'abord pour voir quelles lignes seraient affectees.
05
Delete

DELETE — supprimer des donnees

DELETE — supprimer des lignesSQL
-- Supprimer un enregistrement specifique
DELETE FROM utilisateurs WHERE id = 5;

-- Supprimer selon une condition
DELETE FROM produits WHERE stock = 0;

-- Supprimer toutes les lignes (DANGEREUX — sans WHERE)
-- DELETE FROM utilisateurs;  ← vide toute la table !
-- TRUNCATE TABLE utilisateurs; ← plus rapide, remet AUTO_INCREMENT a 1

-- Verification AVANT suppression (bonne pratique)
-- 1. Tester d'abord avec SELECT
SELECT * FROM produits WHERE stock = 0;
-- 2. Si le resultat est correct, remplacer SELECT * par DELETE FROM
DELETE FROM produits WHERE stock = 0;
!
DELETE sans WHERE supprime tout. Toujours ecrire le WHERE d'abord. Tester avec SELECT avant de lancer DELETE. En production, les suppressions definitives sont souvent remplacees par une suppression logique : UPDATE ... SET actif = 0 (soft delete).
06
Filtres

WHERE — tous les operateurs de filtrage

WHERE — operateurs et combinaisonsSQL
-- Comparaisons
WHERE prix = 19.99       -- egal
WHERE prix != 19.99      -- different
WHERE prix > 10          -- superieur
WHERE prix BETWEEN 10 AND 50  -- intervalle

-- Combinaisons logiques
WHERE stock > 0 AND prix < 100   -- les deux conditions
WHERE role = 'admin' OR role = 'moderateur'
WHERE NOT actif = 1

-- IN : liste de valeurs
WHERE role IN ('admin', 'moderateur')
WHERE id NOT IN (1, 2, 3)

-- NULL
WHERE telephone IS NULL
WHERE telephone IS NOT NULL

-- LIKE : recherche partielle
WHERE nom LIKE 'Alice%'   -- commence par Alice
WHERE nom LIKE '%dupont'  -- finit par dupont
WHERE nom LIKE '%alice%'  -- contient alice

-- GROUP BY : regrouper et agreger
SELECT categorie_id, COUNT(*) AS nb_produits
FROM produits
GROUP BY categorie_id
HAVING COUNT(*) > 5;   -- HAVING = WHERE sur un groupe
07
Relations

JOIN — relier plusieurs tables

Un JOIN combine les donnees de deux tables selon une relation. C'est le fondement du modele relationnel : plutot que de dupliquer les donnees, on les relie par des cles etrangeres.

JOIN — types et exemplesSQL
-- Tables : commandes (id, user_id, total) + utilisateurs (id, nom)

-- INNER JOIN : lignes qui ont une correspondance dans les deux tables
SELECT commandes.id, utilisateurs.nom, commandes.total
FROM commandes
INNER JOIN utilisateurs
    ON commandes.user_id = utilisateurs.id;

-- Avec alias pour raccourcir
SELECT c.id, u.nom, c.total
FROM commandes c
INNER JOIN utilisateurs u ON c.user_id = u.id
WHERE c.total > 100
ORDER BY c.total DESC;

-- LEFT JOIN : toutes les lignes de gauche + correspondances a droite
-- Utile pour lister les utilisateurs MEME sans commande
SELECT u.nom, COUNT(c.id) AS nb_commandes
FROM utilisateurs u
LEFT JOIN commandes c ON u.id = c.user_id
GROUP BY u.id, u.nom;
i
INNER JOIN vs LEFT JOIN. INNER JOIN ne retourne que les lignes qui ont une correspondance dans les deux tables. LEFT JOIN retourne toutes les lignes de la table de gauche, meme si elles n'ont pas de correspondance a droite (les colonnes de droite seront NULL).
OK
Les 5 commandes essentielles : SELECT (lire), INSERT (ajouter), UPDATE (modifier), DELETE (supprimer), JOIN (relier). Maitriser ces 5 + WHERE couvre 90% des besoins d'une application web classique.
