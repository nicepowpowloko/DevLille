Module 27 : Qu'est-ce que le pattern MVC ?

Objectif : Comprendre pourquoi MVC existe, ce que chaque couche fait et comment elles communiquent. C'est la base de tous les frameworks modernes : Symfony, Laravel, CodeIgniter…

Le problème sans MVC — le code spaghetti

Sans architecture : tout dans un seul fichier
// users.php — un désastre : SQL + logique + HTML mélangés $pdo = new PDO("mysql:host=localhost;dbname=test", "root", ""); if ($_POST['action'] === 'delete') { $stmt = $pdo->prepare("DELETE FROM users WHERE id = ?"); $stmt->execute([$_POST['id']]); } $users = $pdo->query("SELECT * FROM users")->fetchAll(); ?><!DOCTYPE html><html><body> <?php foreach($users as $u): ?> <p><?= $u['nom'] ?></p> <!-- XSS : pas de htmlspecialchars ! --> <?php endforeach; ?></body></html>

La solution : 3 couches, 3 responsabilités

Model

  • SQL et BDD
  • Logique métier
  • Validation
  • Pas de HTML
  • Pas de $_GET

View

  • HTML + CSS
  • Affichage seul
  • Reçoit des variables
  • Pas de SQL
  • Pas de logique

Controller

  • Chef d'orchestre
  • Lit $_GET / $_POST
  • Appelle le Model
  • Charge la View
  • Pas de SQL direct

Règle d'or : Le dossier public/ est le seul accessible depuis un navigateur. Tout le reste (app/, core/, config/) est hors portée du web. Vos classes PHP, vos configs et votre BDD sont ainsi protégés.

Module 28 : Le Routeur

Rôle : Le Routeur analyse l'URL (/users, /users/5/edit…) et appelle le bon contrôleur avec les bons paramètres.

Étape 1 — Tout passer par un seul fichier (.htaccess)

.htaccess (à la racine du projet)
Options -MultiViews RewriteEngine On # Si le fichier existe physiquement (CSS, images...) : le servir directement RewriteCond %{REQUEST_FILENAME} !-f RewriteCond %{REQUEST_FILENAME} !-d # Sinon : tout rediriger vers public/index.php RewriteRule ^(.*)$ public/index.php [QSA,L]
public/index.php — Point d'entrée unique
<?php session_start(); declare(strict_types=1); require_once dirname(__DIR__) . '/vendor/autoload.php'; use Core\Router; $url = '/' . trim($_GET['url'] ?? '/', '/'); $method = $_SERVER['REQUEST_METHOD']; $router = new Router(); // ── Déclaration de toutes les routes ─ $router->get('/', 'HomeController', 'index'); $router->get('/users', 'UserController', 'index'); $router->get('/users/create', 'UserController', 'create'); $router->post('/users/store', 'UserController', 'store'); $router->get('/users/{id}', 'UserController', 'show'); $router->get('/users/{id}/edit', 'UserController', 'edit'); $router->post('/users/{id}/update','UserController', 'update'); $router->post('/users/{id}/delete','UserController', 'delete'); $router->dispatch($url, $method);

Étape 2 — La classe Router

core/Router.php
<?php namespace Core; class Router { private array $routes = []; public function get(string $path, string $ctrl, string $action): void { $this->add('GET', $path, $ctrl, $action); } public function post(string $path, string $ctrl, string $action): void { $this->add('POST', $path, $ctrl, $action); } private function add(string $method, string $path, string $ctrl, string $action): void { // Convertit {id} en groupe capturant regex $regex = '#^' . preg_replace('/\{(\w+)\}/', '([^/]+)', $path) . '$#'; $this->routes[] = ['method'=>$method, 'regex'=>$regex, 'ctrl'=>$ctrl, 'action'=>$action]; } public function dispatch(string $url, string $method): void { foreach ($this->routes as $route) { if ($route['method'] !== $method) continue; if (!preg_match($route['regex'], $url, $matches)) continue; array_shift($matches); // supprime le match complet, garde les groupes $class = 'App\\Controller\\' . $route['ctrl']; if (!class_exists($class)) { $this->e404(); return; } $ctrl = new $class(); $action = $route['action']; if (!method_exists($ctrl, $action)) { $this->e404(); return; } $ctrl->$action(...$matches); // passe les paramètres d'URL à la méthode return; } $this->e404(); } private function e404(): void { http_response_code(404); echo '<h1>404 — Page introuvable</h1>'; } }
Exemple concret : URL /users/42/edit → regex #^/users/([^/]+)/edit$# capture "42" → Router appelle UserController::edit("42")

Module 29 : Les Modèles

🎯 Rôle : Le Model est la seule couche qui parle à la base de données. Il contient toutes les requêtes SQL et la logique métier.
config/database.php
<?php // Ajouter ce fichier dans .gitignore — ne jamais commiter les identifiants ! return [ 'host' => 'localhost', 'dbname' => 'ma_base', 'user' => 'root', 'pass' => '', ];
core/Database.php — Singleton PDO
<?php namespace Core; use PDO; class Database { private static ?self $instance = null; private PDO $pdo; private function __construct() { $c = require dirname(__DIR__) . '/config/database.php'; $this->pdo = new PDO( "mysql:host={$c['host']};dbname={$c['dbname']};charset=utf8mb4", $c['user'], $c['pass'], [ PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC, PDO::ATTR_EMULATE_PREPARES => false, ] ); } public static function getInstance(): static { if (static::$instance === null) static::$instance = new static(); return static::$instance; } public function getPdo(): PDO { return $this->pdo; } private function __clone() {} }
core/Model.php — Classe de base avec CRUD générique
<?php namespace Core; use PDO; abstract class Model { protected PDO $pdo; protected string $table; // défini dans chaque modèle enfant public function __construct() { $this->pdo = Database::getInstance()->getPdo(); } public function findAll(string $order = 'id'): array { return $this->pdo->query("SELECT * FROM {$this->table} ORDER BY {$order}")->fetchAll(); } public function findById(int $id): ?array { $s = $this->pdo->prepare("SELECT * FROM {$this->table} WHERE id = ?"); $s->execute([$id]); return $s->fetch() ?: null; } public function delete(int $id): bool { $s = $this->pdo->prepare("DELETE FROM {$this->table} WHERE id = ?"); return $s->execute([$id]); } public function count(): int { return (int) $this->pdo->query("SELECT COUNT(*) FROM {$this->table}")->fetchColumn(); } }
app/Model/UserModel.php
<?php namespace App\Model; use Core\Model; class UserModel extends Model { protected string $table = 'users'; public function findByEmail(string $email): ?array { $s = $this->pdo->prepare("SELECT * FROM users WHERE email = ?"); $s->execute([$email]); return $s->fetch() ?: null; } public function create(array $d): int { $s = $this->pdo->prepare(" INSERT INTO users (nom, email, mot_de_passe) VALUES (:nom, :email, :mdp) "); $s->execute([ 'nom' => htmlspecialchars(trim($d['nom'])), 'email' => strtolower(trim($d['email'])), 'mdp' => password_hash($d['mot_de_passe'], PASSWORD_BCRYPT), ]); return (int) $this->pdo->lastInsertId(); } public function update(int $id, array $d): bool { $s = $this->pdo->prepare("UPDATE users SET nom=:nom, email=:email WHERE id=:id"); return $s->execute([ 'nom' => htmlspecialchars(trim($d['nom'])), 'email' => strtolower(trim($d['email'])), 'id' => $id, ]); } // Validation — retourne un tableau d'erreurs (vide = tout OK) public function valider(array $d): array { $err = []; if (empty(trim($d['nom'] ?? ''))) $err['nom'] = "Le nom est obligatoire"; if (!filter_var($d['email'] ?? '', FILTER_VALIDATE_EMAIL)) $err['email'] = "Email invalide"; elseif ($this->findByEmail($d['email'])) $err['email'] = "Cet email est déjà utilisé"; if (strlen($d['mot_de_passe'] ?? '') < 8) $err['mot_de_passe'] = "8 caractères minimum"; return $err; } }

Module 30 : Les Vues

Règle absolue : Une Vue ne fait que de l'affichage. Pas de SQL, pas de logique métier. Elle reçoit des variables et les intègre dans du HTML.

La classe Controller de base (méthode render)

core/Controller.php
<?php namespace Core; abstract class Controller { // Charge une vue et lui injecte des variables protected function render(string $vue, array $data = []): void { extract($data); // ['users'=>[...]] devient $users = [...] $base = dirname(__DIR__) . '/app/View/'; require $base . 'layout/header.php'; require $base . $vue . '.php'; // ex: "user/index" → View/user/index.php require $base . 'layout/footer.php'; } protected function redirect(string $url): void { header("Location: {$url}"); exit; } // Message flash (succès / erreur affiché une seule fois) protected function flash(string $type, string $msg): void { $_SESSION['flash'] = ['type'=>$type, 'message'=>$msg]; } }

Toujours htmlspecialchars() dans les vues ! C'est la protection de base contre les attaques XSS. Chaque donnée venant de la BDD ou de l'utilisateur doit être échappée avant affichage.

Module 31 : Les Contrôleurs

Rôle : Le Controller est le chef d'orchestre. Il lit la requête HTTP, interroge le Model, et décide quelle Vue afficher — ou vers quelle URL rediriger.
app/Controller/UserController.php — CRUD complet
<?php namespace App\Controller; use Core\Controller; use App\Model\UserModel; class UserController extends Controller { private UserModel $model; public function __construct() { $this->model = new UserModel(); } // GET /users — liste public function index(): void { $this->render('user/index', [ 'title' => 'Utilisateurs', 'users' => $this->model->findAll('nom'), ]); } // GET /users/{id} — détail public function show(string $id): void { $user = $this->model->findById((int) $id); if (!$user) { http_response_code(404); echo '<h1>404</h1>'; return; } $this->render('user/show', ['title' => $user['nom'], 'user' => $user]); } // GET /users/create — formulaire vide public function create(): void { $this->render('user/create', [ 'title' => 'Nouvel utilisateur', 'erreurs' => [], 'ancien' => [], ]); } // POST /users/store — traitement création public function store(): void { $data = [ 'nom' => $_POST['nom'] ?? '', 'email' => $_POST['email'] ?? '', 'mot_de_passe' => $_POST['mot_de_passe'] ?? '', ]; $erreurs = $this->model->valider($data); if (!empty($erreurs)) { // Erreurs → ré-afficher le formulaire avec messages + anciennes valeurs $this->render('user/create', [ 'title' => 'Nouvel utilisateur', 'erreurs' => $erreurs, 'ancien' => $data, ]); return; } $id = $this->model->create($data); $this->flash('success', 'Utilisateur créé !'); $this->redirect("/users/{$id}"); } // GET /users/{id}/edit — formulaire pré-rempli public function edit(string $id): void { $user = $this->model->findById((int) $id); if (!$user) { $this->redirect('/users'); } $this->render('user/edit', [ 'title' => "Modifier {$user['nom']}", 'user' => $user, 'erreurs' => [], ]); } // POST /users/{id}/update — traitement modification public function update(string $id): void { $this->model->update((int) $id, $_POST); $this->flash('success', 'Modifications enregistrées.'); $this->redirect("/users/{$id}"); } // POST /users/{id}/delete — suppression public function delete(string $id): void { $this->model->delete((int) $id); $this->flash('danger', 'Utilisateur supprimé.'); $this->redirect('/users'); } }

Convention de nommage CRUD

Méthode HTTPURLActionRôle
GET/usersindex()Liste tous les éléments
GET/users/{id}show($id)Affiche un élément
GET/users/createcreate()Formulaire de création
POST/users/storestore()Traite et sauvegarde la création
GET/users/{id}/editedit($id)Formulaire de modification
POST/users/{id}/updateupdate($id)Traite et sauvegarde la modification
POST/users/{id}/deletedelete($id)Supprime l'élément
Pourquoi POST pour delete ? Les navigateurs ne font que GET et POST en HTML pur. On utilise donc POST pour toutes les actions qui modifient des données (créer, modifier, supprimer). En JavaScript ou avec une API REST, on utiliserait DELETE et PUT.

Base de données

-- SQL : mini_blog CREATE TABLE users ( id INT AUTO_INCREMENT PRIMARY KEY, nom VARCHAR(100) NOT NULL, email VARCHAR(150) UNIQUE NOT NULL, mot_de_passe VARCHAR(255) NOT NULL, role ENUM('user','admin') DEFAULT 'user' ); CREATE TABLE categories ( id INT AUTO_INCREMENT PRIMARY KEY, nom VARCHAR(80) NOT NULL, slug VARCHAR(80) UNIQUE NOT NULL ); CREATE TABLE articles ( id INT AUTO_INCREMENT PRIMARY KEY, titre VARCHAR(200) NOT NULL, slug VARCHAR(200) UNIQUE NOT NULL, contenu TEXT NOT NULL, publie BOOLEAN DEFAULT 0, category_id INT, user_id INT NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY(category_id) REFERENCES categories(id) ON DELETE SET NULL, FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE );

AuthController — connexion / déconnexion

app/Controller/AuthController.php
<?php namespace App\Controller; use Core\Controller; use App\Model\UserModel; class AuthController extends Controller { // GET /login public function loginForm(): void { $this->render('auth/login', ['title'=>'Connexion', 'erreur'=>null]); } // POST /login public function login(): void { $model = new UserModel(); $user = $model->findByEmail($_POST['email'] ?? ''); if (!$user || !password_verify($_POST['mot_de_passe'] ?? '', $user['mot_de_passe'])) { $this->render('auth/login', [ 'title' => 'Connexion', 'erreur' => 'Email ou mot de passe incorrect', ]); return; } // Connexion réussie : stocker l'utilisateur en session session_regenerate_id(true); // sécurité : évite la fixation de session $_SESSION['user'] = [ 'id' => $user['id'], 'nom' => $user['nom'], 'role' => $user['role'], ]; $this->redirect('/admin'); } // GET /logout public function logout(): void { session_destroy(); $this->redirect('/'); } }

ArticleModel — avec JOIN et slug

app/Model/ArticleModel.php
<?php namespace App\Model; use Core\Model; use PDO; class ArticleModel extends Model { protected string $table = 'articles'; // Articles publiés avec catégorie et auteur (JOIN) public function findPublies(int $limite = 10): array { $s = $this->pdo->prepare(" SELECT a.*, c.nom AS categorie, u.nom AS auteur FROM articles a LEFT JOIN categories c ON a.category_id = c.id LEFT JOIN users u ON a.user_id = u.id WHERE a.publie = 1 ORDER BY a.created_at DESC LIMIT :lim "); $s->bindValue(':lim', $limite, PDO::PARAM_INT); $s->execute(); return $s->fetchAll(); } public function findBySlug(string $slug): ?array { $s = $this->pdo->prepare(" SELECT a.*, c.nom AS categorie, u.nom AS auteur FROM articles a LEFT JOIN categories c ON a.category_id = c.id LEFT JOIN users u ON a.user_id = u.id WHERE a.slug = ? AND a.publie = 1 "); $s->execute([$slug]); return $s->fetch() ?: null; } public function create(array $d, int $userId): int { $s = $this->pdo->prepare(" INSERT INTO articles (titre, slug, contenu, publie, category_id, user_id) VALUES (:titre, :slug, :contenu, :publie, :cat, :uid) "); $s->execute([ 'titre' => htmlspecialchars(trim($d['titre'])), 'slug' => $this->genSlug($d['titre']), 'contenu' => $d['contenu'], 'publie' => isset($d['publie']) ? 1 : 0, 'cat' => $d['category_id'] ?: null, 'uid' => $userId, ]); return (int) $this->pdo->lastInsertId(); } // Génère un slug URL-friendly depuis un titre private function genSlug(string $titre): string { $s = strtolower($titre); $s = str_replace([' ','é','è','ê','à','ç'],['-','e','e','e','a','c'], $s); $s = preg_replace('/[^a-z0-9-]/', '', $s); return trim($s, '-'); } }
