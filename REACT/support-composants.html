Étape 1

C'est quoi un composant ?

Un composant c'est une fonction JavaScript qui retourne du JSX. C'est une brique réutilisable de l'interface — un bouton, une carte, un formulaire, une page entière. On les assemble comme des Lego.

Règle d'or : 1 composant = 1 fichier = 1 responsabilité. Un composant Carte affiche une carte. Pas plus. Si ça fait trop de choses, on découpe.
MonPremierComposant.jsx
// Un composant = une fonction qui retourne du JSX
function Bonjour() {
    return <h1>Bonjour tout le monde !</h1>;
}

// On l'utilise comme une balise HTML
function App() {
    return (
        <div>
            <Bonjour />   // ← composant
            <Bonjour />   // ← réutilisé 3 fois
            <Bonjour />
        </div>
    );
}

export default App; // ← toujours exporter

Étape 3

Les Props — passer des données

Les props c'est le mécanisme pour passer des données du parent vers l'enfant. C'est en lecture seule — un enfant ne peut jamais modifier les props qu'il reçoit.

Props — exemple complet
// ── Composant enfant — reçoit des props ──
function CarteDevs({ nom, poste, salaire, remote = false }) {
    return (
        <div className="carte">
            <h3>{nom}</h3>
            <p>{poste}</p>
            <p>{salaire.toLocaleString('fr-FR')} €</p>
            {remote && <span>🏠 Remote</span>}
        </div>
    );
}

// ── Composant parent — passe des props ──
const devs = [
    { id:1, nom:"Alice",  poste:"Frontend",  salaire:38000, remote:true  },
    { id:2, nom:"Bob",    poste:"Backend",   salaire:44000, remote:false },
    { id:3, nom:"Chloé",  poste:"Fullstack", salaire:52000, remote:true  },
];

function ListeDevs() {
    return (
        <div>
            {devs.map(dev => (
                <CarteDevs
                    key={dev.id}
                    nom={dev.nom}
                    poste={dev.poste}
                    salaire={dev.salaire}
                    remote={dev.remote}
                />
            ))}
        </div>
    );
}
Étape 4

Le State — données qui changent

Le state c'est la mémoire d'un composant. Quand le state change, React re-rend automatiquement le composant. On le crée avec le hook useState.

Variable classique
let count = 0;

function Compteur() {
    return (
        <button onClick={() => count++}>
            {count}
        </button>
    );
}
// count change mais React
// ne re-rend pas → rien ne
// s'affiche !
Avec useState
import { useState } from 'react';

function Compteur() {
    const [count, setCount]
        = useState(0);

    return (
        <button onClick={() => setCount(count + 1)}>
            {count}
        </button>
    );
}
useState — exemples variés
import { useState } from 'react';

function Exemples() {
    // [valeur, fonction pour la changer] = useState(valeur initiale)
    const [count,   setCount]   = useState(0);
    const [nom,     setNom]     = useState("");
    const [visible, setVisible] = useState(true);
    const [devs,    setDevs]    = useState([]);

    return (
        <div>
            {/* Incrémenter */}
            <button onClick={() => setCount(count + 1)}>{count}</button>

            {/* Lier à un input */}
            <input
                value={nom}
                onChange={e => setNom(e.target.value)}
            />
            <p>Bonjour {nom} !</p>

            {/* Afficher / cacher */}
            <button onClick={() => setVisible(!visible)}>Toggle</button>
            {visible && <p>Je suis visible !</p>}
        </div>
    );
}
Ne jamais modifier le state directement : count++ ou devs.push(x) ne déclenchent pas de re-rendu. Toujours passer par setCount / setDevs.
Étape 5

Exemples concrets

// Composant avec filtre en temps réel
import { useState } from 'react';

const devs = [
    { id:1, nom:"Alice",  poste:"Frontend"  },
    { id:2, nom:"Bob",    poste:"Backend"   },
    { id:3, nom:"Chloé",  poste:"Fullstack" },
];

function RechercheDev() {
    const [recherche, setRecherche] = useState("");

    const résultats = devs.filter(dev =>
        dev.nom.toLowerCase().includes(recherche.toLowerCase())
    );

    return (
        <div>
            <input
                placeholder="Rechercher…"
                value={recherche}
                onChange={e => setRecherche(e.target.value)}
            />

            <ul>
                {résultats.map(dev => (
                    <li key={dev.id}>
                        {dev.nom} — {dev.poste}
                    </li>
                ))}
            </ul>
        </div>
    );
}
// Formulaire contrôlé — chaque champ lié au state
import { useState } from 'react';

function FormulaireContact() {
    const [form, setForm] = useState({
        nom: "",
        email: "",
        message: "",
    });

    // Mise à jour générique — fonctionne pour tous les champs
    const handleChange = (e) => {
        setForm({ ...form, [e.target.name]: e.target.value });
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        console.log('Envoyé :', form);
    };

    return (
        <form onSubmit={handleSubmit}>
            <input  name="nom"     value={form.nom}     onChange={handleChange} />
            <input  name="email"   value={form.email}   onChange={handleChange} />
            <textarea name="message" value={form.message} onChange={handleChange} />
            <button type="submit">Envoyer</button>
        </form>
    );
}
// Props + State ensemble — cas réel
import { useState } from 'react';

// Composant enfant — reçoit une prop "label" et gère son propre state
function BoutonCompteur({ label, couleur = "blue" }) {
    const [clics, setClics] = useState(0);

    return (
        <div>
            <button
                style={{ background: couleur }}
                onClick={() => setClics(clics + 1)}
            >
                {label} — cliqué {clics} fois
            </button>
        </div>
    );
}

// Composant parent — passe des props différentes à chaque instance
function App() {
    return (
        <div>
            <BoutonCompteur label="Frontend" couleur="#61dafb" />
            <BoutonCompteur label="Backend"  couleur="#44D62C" />
            <BoutonCompteur label="Fullstack"couleur="#c084fc" />
            {/* Chaque instance a son propre state "clics" */}
        </div>
    );
}
Étape 6

Les erreurs classiques

Pièges fréquents
//  Composant en minuscule — React pense que c'est une balise HTML
function maCarteDevs() { ... }     // ne marche pas
function MaCarteDevs() { ... }     // toujours en majuscule


// Modifier le state directement
const [devs, setDevs] = useState([]);
devs.push({ nom: "Alice" });       //  pas de re-rendu !
setDevs([...devs, { nom: "Alice" }]); // spread + setDevs


// Oublier la prop key dans .map()
devs.map(dev => <CarteDevs nom={dev.nom} />);         // warning
devs.map(dev => <CarteDevs key={dev.id} nom={dev.nom} />); // 


// Passer une fonction comme prop sans la wrapper
<button onClick={handleClick()}>  // s'exécute immédiatement !
<button onClick={handleClick}>   // passe la référence
<button onClick={() => handleClick(id)}> // avec argument
